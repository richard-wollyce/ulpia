# The fleet. See decisions/0011-fleet-layout.md.
#
# Everything under fleet/ is an agent, found by shape rather than declared here,
# so no configuration can be wrong about it. This file is only for what convention
# cannot express.
#
# Attach a base that lives outside the fleet, deliberately:
#   attach = ../work/client-notes
#
# Switch an agent off without deleting it:
#   disable = steve
#
# The orchestrator's identity, read by kb_fleet and handed to whatever model is
# answering. See decisions/0012-naming-and-hosting.md for why this name.

name = Vesta
role = Keeper of the fleet: knows who is here, routes what arrives, and holds the fleet's memory

# The decision records are NOT attached, and that is a decision rather than an omission.
#
# The router asks one question: does this agent know about this subject. An ADR is not an
# agent knowing something, it is the system's own record of how it was built. Attaching it
# put system records in the same index as agent knowledge, and the two answer different
# questions: `decisions/` won the agent fold on 14 of 29 gold questions, and six of that set
# are answerable ONLY from it, so a third of the measurement was about a base that can never
# be the agent who answers.
#
# What Vesta should ask instead: does any agent hold a record about this. A question about
# the system's own design that no agent has written about is a gap, and reporting it as a
# gap is the answer rather than a failure.
#
# The records are still read by people and by an agent that opens them deliberately. They
# are simply not in the retrieval path.
# attach = decisions


# Who answers is a judgement, not a sum. The deterministic score says which files
# share words with the message; it cannot say which agent understands the subject,
# and it cannot say that nobody here does. A classifier reads the roster and the
# evidence and answers both. See decisions/0027.
#
# The contract is a process: dossier on stdin, verdict on stdout. Any model behind
# any runtime satisfies it, and when this line is absent or the command fails, the
# deterministic choice stands, so the fleet never stops routing.
classifier = tools\classify-claude.cmd


# Promotion: how the deposit becomes knowledge. See decisions/0030 and promote.rs.
#
# Two models and they must not be the same one. The first reads inbox/ and proposes;
# it never sees the base. The second decides, three times through three different
# questions, and never sees the first one's reasoning: it is handed the proposal and
# what the deterministic router says the base already holds for those words.
#
# The second is the stronger reader on purpose. A note written badly costs one note.
# A note admitted wrongly costs every question it later wins.
#
# Absent either line, `kb promote` refuses to run rather than degrading into automatic
# extraction, which is the thing it exists to not be.
promoter = tools\promote-claude.cmd
reviewer = tools\review-claude.cmd

# The answerer: prose from what retrieval found, for `kb answer`. After the verdict,
# never inside retrieval (decisions/0018 stands); grounded in the served passages
# only, refusal carried through. Absent this line, `kb answer` prints the reading
# list instead. See decisions/0032.
answerer = tools\answer-claude.cmd
